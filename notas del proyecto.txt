public function index(Request $request)
{
    $user = Auth::user();
    $careers = Carrera::all(); // Retrieve all careers to populate the dropdown
    $maestros = Maestro::all();

    // Retrieve the selected career's ID from the request
    $filtroCarrera = $request->input('career');

    // Retrieve the students based on the selected career
    $formatos = FormatoAnexo19Mensual::when($filtroCarrera, function ($query, $filtroCarrera) {
        return $query->where('NombreCarrera', $filtroCarrera);
    })->get();
    

    if ($user->hasRole('Maestro')) {
        // Si el usuario es un maestro, muestra solo las calificaciones asignadas a él
        $formatos = FormatoAnexo19Mensual::where('NombreMaestro', $user->name)
            ->paginate(10);
    } elseif ($user->hasRole('Alumno')) {
        // Si el usuario es un alumno, muestra solo sus propias formatos
        $formatos = FormatoAnexo19Mensual::where('NombreAlumno', $user->id)
            ->where('NombreAlumno', $user->id)
            ->paginate(10);
    } else {
        // Si el usuario es un administrador, muestra todas las formatos
        $formatos = FormatoAnexo19Mensual::paginate(10);
    }

    return view('FormatoAnexo19Mensual.index', compact('careers', 'formatos', 'filtroCarrera', 'maestros'));
}

    function mostrarAlumnosDeCarrera() {
        var selectedCarrera = document.getElementById("carrera").value;
        var tablaAlumnos = document.getElementById("tablaAlumnos");

        if (selectedCarrera === "") {
            // No se seleccionó ninguna carrera, muestra todos los alumnos
            tablaAlumnos.style.display = "table-row-group";
        } else {
            // Se seleccionó una carrera, realiza la llamada AJAX para obtener los alumnos
            var xhttp = new XMLHttpRequest();
            xhttp.onreadystatechange = function() {
                if (this.readyState === 4 && this.status === 200) {
                    var response = JSON.parse(this.responseText);
                    if (response.success) {
                        // Limpia la tabla de alumnos
                        tablaAlumnos.innerHTML = "";

                        // Agrega los nuevos alumnos a la tabla
                        response.alumnos.forEach(function(alumno) {
                            var row = document.createElement("tr");
                            var cell = document.createElement("td");
                            cell.style.backgroundColor = "white";
                            cell.style.color = "black";
                            cell.style.height = "50px";
                            cell.style.width = "12%";
                            cell.style.border = "2px solid #BABBC2";
                            cell.style.fontFamily = "Century Gothic, sans-serif";
                            cell.style.fontSize = "16px";
                            cell.innerHTML = "<li style='list-style: none;'>" + alumno.Alumno_id + "</li>";
                            row.appendChild(cell);
                            // Agrega otros datos a la fila de la tabla si los tienes

                            tablaAlumnos.appendChild(row);
                        });
                    } else {
                        console.error("Error al obtener los alumnos de la carrera seleccionada.");
                    }
                }
            };
            xhttp.open("GET", "FormatoAnexo19Mensual.index" + selectedCarrera, true);
            xhttp.send();
        }
    }

{

            'salon',
            'formatoanexomensual19.NombreMateria',
            'formatoanexomensual19.Alumno_id'
        )
        ->join('users', 'formatoanexomensual19.NombreAlumno', '=', 'users.id')
        ->join('materias', 'formatoanexomensual19.NombreMateria', '=', 'materias.IdMaterias')
        ->join('carreras', 'formatoanexomensual19.NombreCarrera', '=', 'carreras.IdCarreras');

    $filtroCarrera = $request->input('carrera');
    $filtroMaestro = $request->input('Maestro_id');

    // Filtro de carrera
    if ($filtroCarrera) {
        $formatos->where('carreras.IdCarreras', $filtroCarrera);
    }

    // Filtro de maestro
    if ($filtroMaestro) {
        $formatos->where('formatoanexomensual19.NombreMaestro', $filtroMaestro);
    }

    // Filtrar solo los alumnos asignados al maestro actual (if user is a teacher)
    if ($user->hasRole('Maestro')) {
        $formatos->where('formatoanexomensual19.NombreMaestro', $user->name);
    }

    // Filtrar registros por rol de usuario (if user is an admin, show all maestros)
    if ($user->hasRole('Admin')) {
        $maestros = Maestro::all();
    } else {
        $maestros = Maestro::where('NombreMaestro', $user->name)->get();
    }

    // Apply additional filters if needed
    // ...

    $formatos = $formatos->paginate(10);
    $materias = Materia::all();
    $carreras = Carrera::all();
    $fechaActual = now();

}



public function index(Request $request)
{
    $user = Auth::user();
    $formatos = FormatoAnexo19Mensual::query();
    $filtroCarrera = $request->input('carrera');
    $maestroId = $request->input('Maestro_id');

    // Use aliases for the table names in the JOIN clause
    $formatos->join('carreras', 'formatoanexomensual19.NombreCarrera', '=', 'carreras.IdCarreras');
    

    //filtro de carrera
    if ($filtroCarrera) {
        $formatos->where('carreras.IdCarreras', $filtroCarrera);
    }

    //filtro maestro
    if ($maestroId) {
        $formatos->where('NombreMaestro', $maestroId);
    }

    if ($user->hasRole('Maestro')) {
        // If the user is a teacher, show only the records assigned to them
        $formatos->where('NombreMaestro', $user->name);
    } 

    if ($user->hasRole('Admin')) {
        $maestros = Maestro::all();
    } else {
        $maestros = Maestro::where('NombreMaestro', $user->name)->get();
    }

    if ($filtroCarrera !== null) {
        if ($filtroCarrera === 'todas') {
            // Do nothing as we want all the records for all the carreras
        } elseif (is_array($filtroCarrera)) {
            $formatos->whereIn('carreras.NombreCarrera', $filtroCarrera); // Use the alias in WHERE clause
        } else {
            $formatos->where('carreras.NombreCarrera', $filtroCarrera); // Use the alias in WHERE clause
        }
    }

    $formatos = $formatos->paginate(10);
    $materias = Materia::all();
    $carreras = Carrera::all();
    $maestros = Maestro::all();
    $fechaActual = now();

    return view('FormatoAnexo19Mensual.index', compact('formatos', 'materias', 'carreras', 'maestros', 'fechaActual', 'filtroCarrera', 'maestroId', 'user'));
}
